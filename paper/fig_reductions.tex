  \draw[tgtgreen, very thick] ($(O)+(-0.45,-0.12)$) -- ($(O)+(2.45,0.67)$) node[below right, font=\scriptsize, xshift=-8pt, yshift=-1pt] {$U|\tau\rangle$};
  % harmless vectors and their projections
  \coordinate (p0) at ($(O)+(1.25,0.343)$); \coordinate (t0) at ($(p0)+(0,2.45)$);
  \coordinate (p1) at ($(O)+(2.05,0.562)$); \coordinate (t1) at ($(p1)+(0,1.75)$);
  \draw[->, okblue, thick] (O) -- (t0) node[above, font=\scriptsize] {$\phi_{\mathbf 0}$};
  \draw[->, okblue, thick] (O) -- (t1) node[above right, font=\scriptsize, xshift=-2pt] {$\phi_{\mathbf 1}$};
